cmake_minimum_required(VERSION 3.20)

# CI sets these from the version tag; local builds use the next development version.
set(HLLCALC_VERSION "1.3.0" CACHE STRING "Three-component application version")
set(HLLCALC_VERSION_SUFFIX "" CACHE STRING "Optional prerelease suffix, e.g. rc.1")
if(NOT HLLCALC_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "HLLCALC_VERSION must be MAJOR.MINOR.PATCH without leading zeros")
endif()
set(_version_components "${CMAKE_MATCH_1};${CMAKE_MATCH_2};${CMAKE_MATCH_3}")
foreach(_component IN LISTS _version_components)
    string(LENGTH "${_component}" _length)
    if(_length GREATER 5 OR _component GREATER 65535)
        message(FATAL_ERROR "Version components must fit Windows VERSIONINFO (0..65535)")
    endif()
endforeach()
if(NOT HLLCALC_VERSION_SUFFIX STREQUAL "")
    if(NOT HLLCALC_VERSION_SUFFIX MATCHES "^[0-9A-Za-z-]+(\\.[0-9A-Za-z-]+)*$")
        message(FATAL_ERROR "Invalid prerelease suffix")
    endif()
    string(REPLACE "." ";" _prerelease_identifiers "${HLLCALC_VERSION_SUFFIX}")
    foreach(_identifier IN LISTS _prerelease_identifiers)
        if(_identifier MATCHES "^0[0-9]+$")
            message(FATAL_ERROR "Numeric prerelease identifiers cannot have leading zeros")
        endif()
    endforeach()
endif()

project(HLLCalc VERSION ${HLLCALC_VERSION} DESCRIPTION "Native artillery calculator and crosshair overlay" LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "HLLCalc requires Windows 10 or later")
endif()
enable_language(RC)
include(CTest)
option(HLLCALC_BUILD_UI_TESTS "Build and register intrusive desktop/input tests (never enable on CI)" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(HLLCALC_FULL_VERSION "${PROJECT_VERSION}")
if(NOT HLLCALC_VERSION_SUFFIX STREQUAL "")
    string(APPEND HLLCALC_FULL_VERSION "-${HLLCALC_VERSION_SUFFIX}")
endif()
set(HLLCALC_ICON_FILE "${CMAKE_CURRENT_SOURCE_DIR}/resources/app.ico")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(cmake/version.h.in generated/version.h @ONLY)
configure_file(resources/app.rc.in generated/app.rc @ONLY)

function(hllcalc_defaults target)
    target_include_directories(${target} PRIVATE
        "${PROJECT_SOURCE_DIR}/src" "${PROJECT_SOURCE_DIR}/resources"
        "${PROJECT_BINARY_DIR}/generated")
    if(MSVC)
        target_compile_options(${target} PRIVATE
            $<$<COMPILE_LANGUAGE:CXX>:/utf-8>
            $<$<COMPILE_LANGUAGE:CXX>:/W4>
            $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
    endif()
endfunction()

include(cmake/imgui.cmake)
add_library(hllcalc_i18n STATIC src/i18n/i18n.cpp)
hllcalc_defaults(hllcalc_i18n)
add_library(hllcalc_updater STATIC src/update/updater.cpp src/update/update_core.cpp src/update/helper.cpp)
hllcalc_defaults(hllcalc_updater)
target_link_libraries(hllcalc_updater PUBLIC winhttp bcrypt shell32 ole32 version)
add_library(hllcalc_sharecode STATIC src/sharecode/cs_crosshair.cpp)
hllcalc_defaults(hllcalc_sharecode)
add_library(hllcalc_hotkey STATIC src/input/hotkey.cpp)
hllcalc_defaults(hllcalc_hotkey)
target_link_libraries(hllcalc_hotkey PUBLIC user32 hllcalc_i18n)
add_library(hllcalc_overlay STATIC src/crosshair/crosshair.cpp)
hllcalc_defaults(hllcalc_overlay)
target_link_libraries(hllcalc_overlay PUBLIC user32 gdi32 comdlg32 hllcalc_i18n)

add_executable(HLLCalc WIN32 src/app/main.cpp src/app/ui.cpp "${PROJECT_BINARY_DIR}/generated/app.rc")
hllcalc_defaults(HLLCalc)
target_compile_definitions(HLLCalc PRIVATE UNICODE _UNICODE)
target_link_libraries(HLLCalc PRIVATE imgui hllcalc_sharecode hllcalc_hotkey hllcalc_overlay hllcalc_i18n hllcalc_updater shell32 ole32 uuid)
set_source_files_properties("${PROJECT_BINARY_DIR}/generated/app.rc" PROPERTIES OBJECT_DEPENDS "${HLLCALC_ICON_FILE}")

if(BUILD_TESTING)
    add_executable(ballistics_tests tests/unit/ballistics_tests.cpp)
    add_executable(cs_crosshair_tests tests/unit/cs_crosshair_tests.cpp)
    target_link_libraries(cs_crosshair_tests PRIVATE hllcalc_sharecode)
    add_executable(cs_crosshair_adapter_tests tests/unit/cs_crosshair_adapter_tests.cpp)
    add_executable(hotkey_tests tests/unit/hotkey_tests.cpp)
    target_link_libraries(hotkey_tests PRIVATE hllcalc_hotkey)
    add_executable(motion_tests tests/unit/motion_tests.cpp)
    add_executable(i18n_tests tests/unit/i18n_tests.cpp)
    target_link_libraries(i18n_tests PRIVATE hllcalc_i18n)
    add_executable(update_tests tests/unit/update_tests.cpp src/update/update_core.cpp)
    foreach(_test IN ITEMS ballistics_tests cs_crosshair_tests cs_crosshair_adapter_tests hotkey_tests motion_tests i18n_tests update_tests)
        hllcalc_defaults(${_test})
    endforeach()
    add_test(NAME ballistics COMMAND ballistics_tests)
    add_test(NAME cs_crosshair_decode COMMAND cs_crosshair_tests)
    add_test(NAME cs_crosshair_static_adapter COMMAND cs_crosshair_adapter_tests)
    add_test(NAME nonlinear_motion COMMAND motion_tests)
    add_test(NAME i18n_dictionary COMMAND i18n_tests)
    add_test(NAME update_validation COMMAND update_tests)
    add_test(NAME custom_hotkey COMMAND hotkey_tests)
    set_tests_properties(custom_hotkey PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
    if(HLLCALC_BUILD_UI_TESTS)
        add_executable(ui_tests tests/desktop/ui_tests.cpp)
        hllcalc_defaults(ui_tests)
        target_link_libraries(ui_tests PRIVATE user32)
        add_dependencies(ui_tests HLLCalc)
        add_test(NAME desktop_ui COMMAND ui_tests "$<TARGET_FILE:HLLCalc>")
        add_test(NAME desktop_ui_english COMMAND ui_tests "$<TARGET_FILE:HLLCalc>" --english)
        set_tests_properties(desktop_ui desktop_ui_english PROPERTIES LABELS desktop RUN_SERIAL TRUE TIMEOUT 180
            WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    endif()
endif()
